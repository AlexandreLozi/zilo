// Controlled callback-lifetime characterization only, not a reproduction of the Store scheduler.
ThreadPool.GetMinThreads(out var oldMin, out var minIo);
ThreadPool.GetMaxThreads(out var oldMax, out var maxIo);
if (!ThreadPool.SetMinThreads(1, minIo) || !ThreadPool.SetMaxThreads(1, maxIo)) throw new InvalidOperationException("cannot-bound-threadpool");
try
{
    foreach (var disposeBeforeWorker in new[] { false, true })
    {
        using var workerEntered = new ManualResetEventSlim();
        using var releaseWorker = new ManualResetEventSlim();
        using var source = new CancellationTokenSource();
        var calls = 0;
        source.Token.Register(() => Interlocked.Increment(ref calls));
        var blocker = Task.Run(() => { workerEntered.Set(); if (!releaseWorker.Wait(TimeSpan.FromSeconds(10))) throw new TimeoutException("worker-gate"); });
        Task cancellation;
        try
        {
            if (!workerEntered.Wait(TimeSpan.FromSeconds(5))) throw new TimeoutException("worker-entry");
            cancellation = source.CancelAsync();
            if (disposeBeforeWorker) source.Dispose();
        }
        finally { releaseWorker.Set(); }
        if (!cancellation.Wait(TimeSpan.FromSeconds(5)) || !blocker.Wait(TimeSpan.FromSeconds(5))) throw new TimeoutException("worker-settle");
        Console.WriteLine($"dispose-before-CancelAsync-worker={disposeBeforeWorker}; task-settled={cancellation.IsCompletedSuccessfully}; native-callback-calls={calls}");
    }
}
finally { ThreadPool.SetMaxThreads(oldMax, maxIo); ThreadPool.SetMinThreads(oldMin, minIo); }
